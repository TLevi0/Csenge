using System;
using System.Collections.Generic;
using System.IO;
using System.Text;

namespace ConsoleGyakorlas
{
    internal class Program
    {
        static List<Adat> Lista = new List<Adat>();

        static void Main(string[] args)
        {
            Beolvasas();

            Feladat2();
            Feladat3();
            Feladat4();

            Console.ReadKey();
        }

        private static void Beolvasas()
        {
            StreamReader sr = new StreamReader("adatok.txt", Encoding.UTF8);

            sr.ReadLine(); // fejléc, ha van

            while (!sr.EndOfStream)
            {
                string sor = sr.ReadLine();
                Lista.Add(new Adat(sor));
            }

            sr.Close();
        }
    }
}

internal class Adat
{
    public string Nev { get; set; }
    public int Ev { get; set; }
    public int Darab { get; set; }
    public string Tipus { get; set; }

    public Adat(string sor)
    {
        string[] temp = sor.Split(';');

        Nev = temp[0];
        Ev = int.Parse(temp[1]);
        Darab = int.Parse(temp[2]);
        Tipus = temp[3];
    }
}

private static void Osszeg()
{
    int osszeg = 0;

    for (int i = 0; i < Lista.Count; i++)
    {
        osszeg += Lista[i].Darab;
    }

    Console.WriteLine($"Összeg: {osszeg}");
}

private static void Atlag()
{
    int osszeg = 0;

    for (int i = 0; i < Lista.Count; i++)
    {
        osszeg += Lista[i].Darab;
    }

    double atlag = (double)osszeg / Lista.Count;

    Console.WriteLine($"Átlag: {Math.Round(atlag, 1)}");
}

private static int IdoPercben(string ido)
{
    string[] temp = ido.Split(':');

    int ora = int.Parse(temp[0]);
    int perc = int.Parse(temp[1]);

    return ora * 60 + perc;
}

private static void BelepesKilepes()
{

    string kilepes = "";

        {
            belepes = Lista[i].Ev.ToString();
        }

        if (Lista[i].Nev == azon && Lista[i].Tipus == "kilépés")
        {
            kilepes = Lista[i].Ev.ToString();
        }
    }

    Console.WriteLine($"Első belépés: {belepes}");
    Console.WriteLine($"Utolsó kilépés: {kilepes}");
}

private static void LegalacsonyabbSebesseg()
{
    Console.Write("Adja meg a vizsgált szakasz hosszát km-ben: ");
    double km = double.Parse(Console.ReadLine());

    double meter = km * 1000;

    int min = 999999;

    for (int i = 0; i < Lista.Count; i++)
    {
        if (Lista[i].Darab <= meter && int.TryParse(Lista[i].Tipus, out int sebesseg))
        {
            if (sebesseg < min)
            {
                min = sebesseg;
            }
        }
    }

    Console.WriteLine($"Az első {km} km-en {min} km/h volt a legalacsonyabb megengedett sebesség.");
}

Átírás:
Lista[i].Darab = távolság méterben
Lista[i].Tipus = jelzés, ami lehet szám vagy szöveg

private static void Ismetlodo()
{
    for (int i = 0; i < Lista.Count; i++)
    {
        bool voltMar = false;

        for (int k = 0; k < i; k++)
        {
            if (Lista[k].Nev == Lista[i].Nev)
            {
                voltMar = true;
            }
        }

        if (!voltMar)
        {
            int db = 0;

            for (int j = 0; j < Lista.Count; j++)
            {
                if (Lista[j].Nev == Lista[i].Nev)
                {
                    db++;
                }
            }

            if (db >= 2)
            {
                Console.WriteLine($"{Lista[i].Nev} - {db} db");
            }
        }
    }
}
