public class Termek
{
    public string Nev { get; set; }
    public int Ev { get; set; }
    public int Darab { get; set; }
    public string Tipus { get; set; }

    public Termek(string nev, int ev, int darab, string tipus)
    {
        Nev = nev;
        Ev = ev;
        Darab = darab;
        Tipus = tipus;
    }
}

BEOLVASÁS:

static List<Termek> Termekek = new List<Termek>();

static void BeolvasasTobbSoros()
{
    string[] sorok = File.ReadAllLines("adatok.txt");

    for (int i = 0; i < sorok.Length; i += 4)
    {
        string nev = sorok[i];
        int ev = int.Parse(sorok[i + 1]);
        int darab = int.Parse(sorok[i + 2]);
        string tipus = sorok[i + 3];

        Termekek.Add(new Termek(nev, ev, darab, tipus));
    }
}

public class TermekAlap
{
    public int Id { get; set; }
    public string Nev { get; set; }
    public string Tipus { get; set; }

    public TermekAlap(string sor)
    {
        string[] t = sor.Split(';');

        Id = int.Parse(t[0]);
        Nev = t[1];
        Tipus = t[2];
    }
}

public class Eladas
{
    public int TermekId { get; set; }
    public int Ev { get; set; }
    public int Darab { get; set; }

    public Eladas(string sor)
    {
        string[] t = sor.Split(';');

        TermekId = int.Parse(t[0]);
        Ev = int.Parse(t[1]);
        Darab = int.Parse(t[2]);
    }
}

BEOLVASÁS:

static List<TermekAlap> TermekAlapok = new List<TermekAlap>();
static List<Eladas> Eladasok = new List<Eladas>();

static void BeolvasasTobbFajl()
{
    TermekAlapok = File.ReadAllLines("termekek.txt")
                       .Skip(1)
                       .Select(x => new TermekAlap(x))
                       .ToList();

    Eladasok = File.ReadAllLines("eladasok.txt")
                   .Skip(1)
                   .Select(x => new Eladas(x))
                   .ToList();
}

JOIN LEKÉRDEZÉS:

static void JoinPelda()
{
    var eredmeny = Eladasok
        .Join(
            TermekAlapok,
            eladas => eladas.TermekId,
            termek => termek.Id,
            (eladas, termek) => new
            {
                termek.Nev,
                termek.Tipus,
                eladas.Ev,
                eladas.Darab
            }
        )
        .ToList();

    foreach (var x in eredmeny)
    {
        Console.WriteLine($"{x.Nev} {x.Tipus} {x.Ev} {x.Darab}");
    }
}

LÉNYEG:
Join akkor kell, ha egyik fájlban csak ID van, másik fájlban a név.

static void TipusonkentOsszeg()
{
    var eredmeny = Termekek
        .GroupBy(x => x.Tipus)
        .Select(g => new
        {
            Tipus = g.Key,
            Osszeg = g.Sum(x => x.Darab),
            Darab = g.Count(),
            Atlag = g.Average(x => x.Darab)
        })
        .OrderByDescending(x => x.Osszeg)
        .ToList();

    foreach (var x in eredmeny)
    {
        Console.WriteLine($"{x.Tipus}: {x.Osszeg} összesen, {x.Darab} db, átlag: {Math.Round(x.Atlag, 1)}");
    }
}

LÉNYEG:
g.Key = csoport neve
g.Sum(...) = csoporton belüli összeg

public class Belepes
{
    public string Kod { get; set; }
    public TimeSpan Be { get; set; }
    public TimeSpan Ki { get; set; }

    public Belepes(string sor)
    {
        string[] t = sor.Split(';');

        Kod = t[0];
        Be = TimeSpan.Parse(t[1]);
        Ki = TimeSpan.Parse(t[2]);
    }

    public int BentToltottPerc
    {
        get { return (int)(Ki - Be).TotalMinutes; }
    }
}

LEKÉRDEZÉS:

static List<Belepes> Belepesek = new List<Belepes>();

static void BentToltottIdo()
{
    var eredmeny = Belepesek
        .OrderByDescending(x => x.BentToltottPerc)
        .ToList();

    foreach (var x in eredmeny)
    {
        Console.WriteLine($"{x.Kod}: {x.BentToltottPerc} perc");
    }
}

24) SAFE PARSE – HA ROSSZ ADAT IS LEHET

public class BiztonsagosAdat
{
    public string Nev { get; set; }
    public int Ev { get; set; }
    public int Darab { get; set; }

    public BiztonsagosAdat(string sor)
    {
        string[] t = sor.Split(';');

        Nev = t[0];

        if (!int.TryParse(t[1], out int ev))
        {
            ev = 0;
        }

        if (!int.TryParse(t[2], out int darab))
        {
            darab = 0;
        }

        Ev = ev;
        Darab = darab;
    }
}

static void MindenEvbenSzerepelt()
{
    var evek = Termekek
        .Select(x => x.Ev)
        .Distinct()
        .ToList();

    var eredmeny = Termekek
        .GroupBy(x => x.Nev)
        .Where(g => evek.All(ev => g.Any(x => x.Ev == ev)))
        .Select(g => g.Key)
        .ToList();

    foreach (string nev in eredmeny)
    {
        Console.WriteLine(nev);
    }
}

static void NovekedesKetEvKozott()
{
    var eredmeny = Termekek
        .GroupBy(x => x.Nev)
        .Select(g => new
        {
            Nev = g.Key,
            ElsoEv = g.Where(x => x.Ev == 2023).Sum(x => x.Darab),
            MasodikEv = g.Where(x => x.Ev == 2024).Sum(x => x.Darab)
        })
        .Where(x => x.MasodikEv > x.ElsoEv)
        .Select(x => new
        {
            x.Nev,
            Novekedes = x.MasodikEv - x.ElsoEv
        })
        .OrderByDescending(x => x.Novekedes)
        .ToList();

    foreach (var x in eredmeny)
    {
        Console.WriteLine($"{x.Nev}: +{x.Novekedes}");
    }
}

static void HianyzoEvek()
{
    var osszesEv = Termekek
        .Select(x => x.Ev)
        .Distinct()
        .OrderBy(x => x)
        .ToList();

    var eredmeny = Termekek
        .GroupBy(x => x.Nev)
        .Select(g => new
        {
            Nev = g.Key,
            HianyzoEvek = osszesEv
                .Where(ev => !g.Any(x => x.Ev == ev))
                .ToList()
        })
        .Where(x => x.HianyzoEvek.Any())
        .ToList();

    foreach (var x in eredmeny)
    {
        Console.WriteLine($"{x.Nev}: {string.Join(", ", x.HianyzoEvek)}");
    }
}

34) MINI TELJES PROGRAM VÁZ NEHÉZ LINQ-HOZ
============================================================

using System;
using System.Collections.Generic;
using System.IO;
using System.Linq;

namespace ConsoleNehez
{
    internal class Program
    {
        static List<Termek> Termekek = new List<Termek>();

        static void Main(string[] args)
        {
            Beolvasas();

            TipusonkentOsszeg();
            EvenkentiStatisztika();
            EvTipusSzerint();
            Top3();
            MindenEvbenSzerepelt();

            Console.ReadKey();
        }

        static void Beolvasas()
        {
            Termekek = File.ReadAllLines("adatok.txt")
                .Skip(1)
                .Where(x => x.Trim() != "")
                .Select(x =>
                {
                    string[] t = x.Split(';');

                    return new Termek(
                        t[0],
                        int.Parse(t[1]),
                        int.Parse(t[2]),
                        t[3]
                    );
                })
                .ToList();
        }

        static void TipusonkentOsszeg()
        {
            var eredmeny = Termekek
                .GroupBy(x => x.Tipus)
                .Select(g => new
                {
                    Tipus = g.Key,
                    Osszeg = g.Sum(x => x.Darab)
                })
                .OrderByDescending(x => x.Osszeg)
                .ToList();

            foreach (var x in eredmeny)
            {
                Console.WriteLine($"{x.Tipus}: {x.Osszeg}");
            }
        }

        static void EvenkentiStatisztika()
        {
            var eredmeny = Termekek
                .GroupBy(x => x.Ev)
                .Select(g => new
                {
                    Ev = g.Key,
                    Db = g.Count(),
                    Osszeg = g.Sum(x => x.Darab)
                })
                .OrderBy(x => x.Ev)
                .ToList();

            foreach (var x in eredmeny)
            {
                Console.WriteLine($"{x.Ev}: {x.Db} db, {x.Osszeg} összesen");
            }
        }

        static void EvTipusSzerint()
        {
            var eredmeny = Termekek
                .GroupBy(x => new { x.Ev, x.Tipus })
                .Select(g => new
                {
                    g.Key.Ev,
                    g.Key.Tipus,
                    Osszeg = g.Sum(x => x.Darab)
                })
                .OrderBy(x => x.Ev)
                .ThenBy(x => x.Tipus)
                .ToList();

            foreach (var x in eredmeny)
            {
                Console.WriteLine($"{x.Ev} {x.Tipus}: {x.Osszeg}");
            }
        }

        static void Top3()
        {
            var top3 = Termekek
                .OrderByDescending(x => x.Darab)
                .Take(3)
                .ToList();

            foreach (var x in top3)
            {
                Console.WriteLine($"{x.Nev}: {x.Darab}");
            }
        }

        static void MindenEvbenSzerepelt()
        {
            var evek = Termekek
                .Select(x => x.Ev)
                .Distinct()
                .ToList();

            var eredmeny = Termekek
                .GroupBy(x => x.Nev)
                .Where(g => evek.All(ev => g.Any(x => x.Ev == ev)))
                .Select(g => g.Key)
                .ToList();

            foreach (string nev in eredmeny)
            {
                Console.WriteLine($"{nev} minden évben szerepelt.");
            }
        }
    }

    public class Termek
    {
        public string Nev { get; set; }
        public int Ev { get; set; }
        public int Darab { get; set; }
        public string Tipus { get; set; }

        public Termek(string nev, int ev, int darab, string tipus)
        {
            Nev = nev;
            Ev = ev;
            Darab = darab;
            Tipus = tipus;
        }
    }
}
