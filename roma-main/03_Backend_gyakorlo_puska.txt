============================================================
1) PROGRAM.CS – ALAP WEB API
============================================================

using Microsoft.EntityFrameworkCore;
using System.Text.Json.Serialization;
using VizsgaBackend.Models;

var builder = WebApplication.CreateBuilder(args);

builder.Services.AddControllers()
    .AddJsonOptions(x =>
        x.JsonSerializerOptions.ReferenceHandler = ReferenceHandler.IgnoreCycles);

builder.Services.AddEndpointsApiExplorer();
builder.Services.AddSwaggerGen();

builder.Services.AddDbContext<AppDbContext>(options =>
{
    string connectionString = builder.Configuration.GetConnectionString("DefaultConnection");
    options.UseMySql(connectionString, ServerVersion.AutoDetect(connectionString));
});

builder.Services.AddCors(options =>
{
    options.AddPolicy("frontend", policy =>
    {
        policy.WithOrigins("http://localhost:5173")
              .AllowAnyHeader()
              .AllowAnyMethod();
    });
});

var app = builder.Build();

app.UseSwagger();
app.UseSwaggerUI();

app.UseCors("frontend");

app.UseAuthorization();

app.MapControllers();

app.Run();

22) TELJES FILMSCONTROLLER MINTA

using Microsoft.AspNetCore.Mvc;
using Microsoft.EntityFrameworkCore;
using VizsgaBackend.Models;

namespace VizsgaBackend.Controllers
{
    [Route("api/[controller]")]
    [ApiController]
    public class FilmsController : ControllerBase
    {
        private readonly AppDbContext _context;
        private readonly IConfiguration _configuration;

        public FilmsController(AppDbContext context, IConfiguration configuration)
        {
            _context = context;
            _configuration = configuration;
        }

        [HttpGet]
        public IActionResult GetAll()
        {
            var films = _context.Films.ToList();
            return Ok(films);
        }

        [HttpGet("lista")]
        public IActionResult GetList()
        {
            var films = _context.Films
                .Include(x => x.Actor)
                .Include(x => x.FilmType)
                .Select(x => new
                {
                    x.Id,
                    x.Title,
                    x.ReleaseDate,
                    ActorName = x.Actor.Name,
                    TypeName = x.FilmType.Name
                })
                .ToList();

            return Ok(films);
        }

        [HttpGet("{id}")]
        public IActionResult GetById(int id)
        {
            var film = _context.Films.FirstOrDefault(x => x.Id == id);

            if (film == null)
            {
                return NotFound();
            }

            return Ok(film);
        }

        [HttpPost]
        public IActionResult Create(Film film)
        {
            _context.Films.Add(film);
            _context.SaveChanges();

            return Ok(film);
        }

        [HttpPut("{id}")]
        public IActionResult Update(int id, Film ujFilm)
        {
            var film = _context.Films.FirstOrDefault(x => x.Id == id);

            if (film == null)
            {
                return NotFound();
            }

            film.Title = ujFilm.Title;
            film.ReleaseDate = ujFilm.ReleaseDate;
            film.ActorId = ujFilm.ActorId;
            film.FilmTypeId = ujFilm.FilmTypeId;

            _context.SaveChanges();

            return Ok(film);
        }

        [HttpDelete("{id}")]
        public IActionResult Delete(int id)
        {
            var film = _context.Films.FirstOrDefault(x => x.Id == id);

            if (film == null)
            {
                return NotFound();
            }

            _context.Films.Remove(film);
            _context.SaveChanges();

            return Ok();
        }
    }
}
