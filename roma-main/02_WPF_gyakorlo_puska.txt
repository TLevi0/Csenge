using System;
using System.Collections.Generic;
using System.Linq;
using System.Windows;
using System.Windows.Controls;

namespace WPFgyakorlas
{
    public partial class MainWindow : Window
    {
        private List<AutoAdatok> EredetiLista = new List<AutoAdatok>();
        private List<AutoAdatok> AktualLista = new List<AutoAdatok>();

        public MainWindow()
        {
            InitializeComponent();
        }

        private void Window_Loaded(object sender, RoutedEventArgs e)
        {
            Program.Beolvasas();

            EredetiLista = Program.AdatokLista.ToList();
            AktualLista = Program.AdatokLista.ToList();

            DTGadatok.ItemsSource = AktualLista;

            CBRendezes.Items.Add("Név");
            CBRendezes.Items.Add("Év");
            CBRendezes.Items.Add("Darab");

            OsszegzesKiir();
        }

        private void Frissit()
        {
            DTGadatok.ItemsSource = null;
            DTGadatok.ItemsSource = AktualLista;

            OsszegzesKiir();
        }

        private void OsszegzesKiir()
        {
            int db = AktualLista.Count;
            int osszeg = 0;

            for (int i = 0; i < AktualLista.Count; i++)
            {
                osszeg += AktualLista[i].Darab;
            }

            double atlag = 0;

            if (db > 0)
            {
                atlag = (double)osszeg / db;
            }

            TBOsszegzes.Text = $"Darab: {db}, összeg: {osszeg}, átlag: {Math.Round(atlag, 1)}";
        }
    }
}

private void CBRendezes_SelectionChanged(object sender, SelectionChangedEventArgs e)
{
    if (CBRendezes.SelectedItem == null)
    {
        return;
    }

    string valasztott = CBRendezes.SelectedItem.ToString();

    if (valasztott == "Név")
    {
        AktualLista = AktualLista.OrderBy(x => x.Nev).ToList();
    }
    else if (valasztott == "Év")
    {
        AktualLista = AktualLista.OrderBy(x => x.Ev).ToList();
    }
    else if (valasztott == "Darab")
    {
        AktualLista = AktualLista.OrderBy(x => x.Darab).ToList();
    }

    Frissit();
}

private void BTSzures_Click(object sender, RoutedEventArgs e)
{
    string keresett = TBSzures.Text.ToLower();

    AktualLista = new List<AutoAdatok>();

    for (int i = 0; i < EredetiLista.Count; i++)
    {
        if (EredetiLista[i].Tipus.ToLower().Contains(keresett))
        {
            AktualLista.Add(EredetiLista[i]);
        }
    }

    Frissit();
}

8) PROGRAM.CS BEOLVASÁS WPF-HEZ
============================================================

using System.Collections.Generic;
using System.IO;
using System.Text;

namespace WPFgyakorlas
{
    internal class Program
    {
        public static List<AutoAdatok> AdatokLista = new List<AutoAdatok>();

        public static void Beolvasas()
        {
            StreamReader sr = new StreamReader("adatok.txt", Encoding.UTF8);

            sr.ReadLine();

            while (!sr.EndOfStream)
            {
                string sor = sr.ReadLine();
                AdatokLista.Add(new AutoAdatok(sor));
            }

            sr.Close();
        }
    }
}

9) ADATOSZTÁLY WPF-HEZ

namespace WPFgyakorlas
{
    public class AutoAdatok
    {
        public string Nev { get; set; }
        public int Ev { get; set; }
        public int Darab { get; set; }
        public string Tipus { get; set; }

        public AutoAdatok(string sor)
        {
            string[] temp = sor.Split(';');

            Nev = temp[0];
            Ev = int.Parse(temp[1]);
            Darab = int.Parse(temp[2]);
            Tipus = temp[3];
        }
    }
}

private void TipusokBetoltese()
{
    CBTipus.Items.Clear();

    for (int i = 0; i < EredetiLista.Count; i++)
    {
        if (!CBTipus.Items.Contains(EredetiLista[i].Tipus))
        {
            CBTipus.Items.Add(EredetiLista[i].Tipus);
        }
    }
}

private void BTFajlbaIras_Click(object sender, RoutedEventArgs e)
