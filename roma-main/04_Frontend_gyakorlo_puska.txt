7) FILMEK.JSX – GET LISTA + TÖRLÉS
============================================================

import { useState, useEffect } from 'react'
import { Link } from 'react-router-dom'
import axios from 'axios'

function Filmek() {
  const [filmek, setFilmek] = useState([])

  const betoltes = () => {
    axios.get('https://localhost:7232/api/films')
      .then(res => setFilmek(res.data))
      .catch(err => console.log(err))
  }

  useEffect(() => {
    betoltes()
  }, [])

  const torles = (id) => {
    axios.delete(`https://localhost:7232/api/films/${id}`)
      .then(() => betoltes())
      .catch(err => console.log(err))
  }

  return (
    <>
      <h1>Filmek</h1>

      <table className="table table-striped">
        <thead>
          <tr>
            <th>Cím</th>
            <th>Dátum</th>
            <th>Művelet</th>
          </tr>
        </thead>

        <tbody>
          {filmek.map((film, index) => (
            <tr key={index}>
              <td>{film.title}</td>
              <td>{film.releaseDate}</td>
              <td>
                <Link className="btn btn-primary btn-sm me-2" to={`/filmek/${film.id}`}>
                  Részletek
                </Link>

                <Link className="btn btn-warning btn-sm me-2" to={`/modosit/${film.id}`}>
                  Módosítás
                </Link>

                <button className="btn btn-danger btn-sm" onClick={() => torles(film.id)}>
                  Törlés
                </button>
              </td>
            </tr>
          ))}
        </tbody>
      </table>
    </>
  )
}

export default Filmek

11) MÓDOSÍTÁS – GET + PUT
============================================================

import { useState, useEffect } from 'react'
import { useParams, useNavigate } from 'react-router-dom'
import axios from 'axios'

function ModositFilm() {
  const { id } = useParams()
  const navigate = useNavigate()

  const [title, setTitle] = useState('')
  const [releaseDate, setReleaseDate] = useState('')
  const [actorId, setActorId] = useState('')
  const [filmTypeId, setFilmTypeId] = useState('')

  useEffect(() => {
    axios.get(`https://localhost:7232/api/films/${id}`)
      .then(res => {
        setTitle(res.data.title)
        setReleaseDate(res.data.releaseDate)
        setActorId(res.data.actorId)
        setFilmTypeId(res.data.filmTypeId)
      })
      .catch(err => console.log(err))
  }, [id])

  const kuldes = (e) => {
    e.preventDefault()

    const modositottFilm = {
      title,
      releaseDate,
      actorId: Number(actorId),
      filmTypeId: Number(filmTypeId)
    }

    axios.put(`https://localhost:7232/api/films/${id}`, modositottFilm)
      .then(() => navigate('/filmek'))
      .catch(err => console.log(err))
  }

  return (
    <>
      <h1>Film módosítása</h1>

      <form onSubmit={kuldes}>
        <input className="form-control mb-2" value={title} onChange={e => setTitle(e.target.value)} />
        <input className="form-control mb-2" type="date" value={releaseDate} onChange={e => setReleaseDate(e.target.value)} />
        <input className="form-control mb-2" type="number" value={actorId} onChange={e => setActorId(e.target.value)} />
        <input className="form-control mb-2" type="number" value={filmTypeId} onChange={e => setFilmTypeId(e.target.value)} />

        <button className="btn btn-warning" type="submit">
          Módosítás
        </button>
      </form>
    </>
  )
}

export default ModositFilm

19) GYAKORI HIBÁK
============================================================

Semmit nem ír ki:
map-ben nincs return.

Rossz:
{filmek.map((film) => {
  <p>{film.title}</p>
})}

Jó:
{filmek.map((film) => (
  <p>{film.title}</p>
))}

Vagy:
{filmek.map((film) => {
  return <p>{film.title}</p>
})}

POST nem működik:
- endpoint rossz
- CORS nincs backendben
- Number(actorId) kimaradt
- mezőnév nem egyezik
- dátum nem jó formátum
- backend nem fut

Route nem működik:
main.jsx-ben kell BrowserRouter
App.jsx-ben kell Routes és Route

useParams nem működik:
import { useParams } from 'react-router-dom'

Link nem működik:
import { Link } from 'react-router-dom'

Bootstrap nem működik:
main.jsx:
import 'bootstrap/dist/css/bootstrap.min.css'
