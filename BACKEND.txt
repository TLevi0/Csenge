        [HttpGet("ById/{nev}")]
        public IActionResult ById(string nev)
        {
            try
            {
                using (var cx = new CsatahajokContext())
                {
                    var response = cx.Hajos.FirstOrDefault(h => h.Nev == nev);

                    if (response != null)
                    {
                        return Ok(response);
                    }
                    else
                    {
                        return StatusCode(404, $"Nincs ilyen nevű hajó: {nev}");
                    }
                }
            }
            catch (Exception ex)
            {
                return StatusCode(400, ex.Message);
            }
        }
    }

{

        }
